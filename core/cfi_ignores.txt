# The core calls the app through pointers it gets from the app (BC_APP_ENTRY, bc_host), in another
# library, which Clang's indirect-call check traps on in official Linux builds. This drops the
# check in the core's own functions only; they keep their type data, so Chromium's checked calls
# into them still pass, which -fno-sanitize=cfi-icall on the core's files would break.
[cfi-icall]
fun:*browser_core*
fun:bc_*
